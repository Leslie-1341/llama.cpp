drop view if exists customer_account_number;
create sql security invoker view customer_account_number as
select c.id as customer_id,
 c.utility_id as utility_id,
 c.util_internal_id as account_number,
 cs.search_util_internal_id as search_account_number
-- c.util_internal_id_2 as account_number
 from customer c inner join customer_search cs on cs.customer_id=c.id;
